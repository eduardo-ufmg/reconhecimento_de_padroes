\section{Implementação}

\begin{frame}
\begin{itemize}
    \item $ \text{support}(f) = \frac{1}{|C(f)|}\sum_{\text{kNN}\in{}C(f)}\text{acc}(\text{kNN}) $
    \item $ \text{number\_of\_iterations\_stg1} = \lfloor \ln(4/\text{current\_number\_of\_features}) / \ln(1-\text{elimination\_rate}) \rfloor $
    \item $ \text{number\_of\_iterations\_stg2} = \lfloor (\text{current\_number\_of\_features}-4)/\text{elimination\_const} \rfloor $
    \item a cada iteração, computa $ \text{support}(f) \forall f $, mantém os \textit{p} melhores e atualiza \textit{p}
\end{itemize}
\end{frame}
