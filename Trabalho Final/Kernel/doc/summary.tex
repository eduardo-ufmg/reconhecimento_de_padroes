\section*{1. Initial Kernel Parameter Estimation Procedure}
\subsection*{Inputs}
\begin{itemize}
    \item Data matrix $\mat{X} \in \mathbb{R}^{N \times D}$.
    \item Estimation type: 'cov' or 'scale'.
\end{itemize}
\subsection*{Outputs}
\begin{itemize}
    \item Normalization factor $\eta \in \mathbb{R}$.
    \item Inverse matrix $\mat{\Psi}^{-1} \in \mathbb{R}^{D \times D}$.
\end{itemize}

\subsection*{Procedure A: Covariance-based Estimation ('cov')}
\begin{enumerate}
    \item \textbf{Empirical Covariance Matrix ($\mat{\Sigma}$):}
    Let $\hat{\vecbf{\mu}} = \frac{1}{N} \sum_{i=1}^{N} \vecbf{x}_i$.
    $$ \mat{\Sigma} = \frac{1}{N-1} \sum_{i=1}^{N} (\vecbf{x}_i - \hat{\vecbf{\mu}})^T (\vecbf{x}_i - \hat{\vecbf{\mu}}) $$
    \item \textbf{Covariance Matrix Validation:} Check if $\det(\mat{\Sigma})$ is non-negligible and positive.
    \item \textbf{Normalization Factor ($\eta$):}
    $$ \eta = \frac{1}{(2\pi)^{D/2} \sqrt{\det(\mat{\Sigma})}} $$
    \item \textbf{Inverse Covariance Matrix ($\mat{\Psi}^{-1} = \mat{\Sigma}^{-1}$):} $\mat{\Psi}^{-1} = \mat{\Sigma}^{-1}$.
\end{enumerate}

\subsection*{Procedure B: Scale-based Estimation ('scale')}
\begin{enumerate}
    \item \textbf{Inverse Matrix ($\mat{\Psi}^{-1} = \mat{I}$):} $\mat{\Psi}^{-1} = \mat{I}$ (identity matrix).
    \item \textbf{Normalization Factor ($\eta$):}
    $$ \eta = \frac{1}{(2\pi)^{D/2}} $$
\end{enumerate}

\section*{2. Parameterized Kernel Function}
\subsection*{Inputs}
\begin{itemize}
    \item Data matrix $\mat{X} \in \mathbb{R}^{N_X \times D}$.
    \item Data matrix $\mat{Y} \in \mathbb{R}^{N_Y \times D}$.
    \item Initial normalization factor $\eta_0 \in \mathbb{R}$.
    \item Initial inverse matrix $\mat{\Psi}^{-1}_0 \in \mathbb{R}^{D \times D}$.
    \item Bandwidth hyperparameter $h \in \mathbb{R}$, $h \neq 0$.
\end{itemize}
\subsection*{Output}
\begin{itemize}
    \item Normalized kernel matrix $\mat{K} \in \mathbb{R}^{N_X \times N_Y}$.
\end{itemize}

\subsection*{Calculation Steps}
\begin{enumerate}
    \item \textbf{Pre-computation:}
    \begin{itemize}
        \item Scaled Inverse Matrix: $\mat{\Psi}^{-1}_h = \frac{\mat{\Psi}^{-1}_0}{h^2}$.
        \item Scaled Normalization Factor: $\eta_h = \frac{\eta_0}{h^D}$ (with boundary conditions for $\eta_0=0$ or very small $h^D$).
    \end{itemize}
    \item \textbf{Quadratic Form Matrix ($\mat{Q}$):} $Q_{ij} = (\vecbf{x}_i - \vecbf{y}_j)^T \mat{\Psi}^{-1}_h (\vecbf{x}_i - \vecbf{y}_j)$.
    Expanded as:
    $$ Q_{ij} = (\text{diag}(\mat{X} \mat{\Psi}^{-1}_h \mat{X}^T))_i - 2 (\mat{X} \mat{\Psi}^{-1}_h \mat{Y}^T)_{ij} + (\text{diag}(\mat{Y} \mat{\Psi}^{-1}_h \mat{Y}^T))_j $$
    \item \textbf{Unnormalized Kernel-like Values ($\mat{P}$):}
    $$ P_{ij} = \eta_h \exp\left(-\frac{1}{2} Q_{ij}\right) $$
    \item \textbf{Final Normalization:}
    Let $P_{max} = \max_{i,j} (P_{ij})$.
    $$ K_{ij} = \begin{cases} 0 & \text{if } P_{max} \approx 0 \\ \frac{P_{ij}}{P_{max}} & \text{otherwise} \end{cases} $$
\end{enumerate}
