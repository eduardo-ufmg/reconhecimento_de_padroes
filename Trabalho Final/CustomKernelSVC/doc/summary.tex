\section*{Custom Kernel Support Vector Classifier}

\subsection*{1. Initialization Parameters}
\begin{itemize}
    \item $h_{bounds} = (h_{min}, h_{max})$: Search interval for kernel hyperparameter $h$.
    \item $\text{kernel\_fit\_type}$: String specifying method for initial kernel parameter estimation.
    \item $\text{objective\_metric}$: String ('spatial' or 'axis') selecting the objective function for $h$ optimization.
    \item $\text{svm\_kwargs}$: Arguments for the underlying \texttt{sklearn.svm.SVC}.
\end{itemize}

\subsection*{2. Hyperparameter Optimization Objective Criterion}
This criterion evaluates a candidate kernel hyperparameter $h_{candidate}$.

\paragraph{Inputs:}
\begin{itemize}
    \item $h_{candidate}$: Candidate value for $h$.
    \item $\mat{X} \in \mathbb{R}^{N \times D}$: Training data matrix ($N$ samples, $D$ features).
    \item $\vecbf{y} \in \mathcal{Y}^N$: Training class labels. Let $\mathcal{C} = \{C_0, C_1, \dots, C_{K-1}\}$ be $K$ unique classes.
\end{itemize}

\paragraph{Pre-requisites (from \texttt{fit} method):}
\begin{itemize}
    \item $\eta \in \mathbb{R}$, $\mat{\Sigma}^{-1}$: Initial parameters from $\text{kernel\_fit}(\mat{X})$.
\end{itemize}

\paragraph{Calculation Steps:}
\begin{enumerate}
    \item \textbf{Parameterized Kernel Matrix Computation:}
    Compute $\mat{K}_h \in \mathbb{R}^{N \times N}$ where $(\mat{K}_h)_{uv} = \text{kernel}(\vecbf{x}_u, \vecbf{x}_v; \eta, \mat{\Sigma}^{-1}, h_{candidate})$.
    The function $\text{kernel}(\cdot)$ is an external procedure.

    \item \textbf{Data Projection Vector Computation:} For $i=1, \dots, N$:
    \begin{align*}
        (\vecbf{Q}_{C_0}(h))_i &= \sum_{j \text{ s.t. } y_j = C_0} (\mat{K}_h)_{i,j} \\
        (\vecbf{Q}_{C_1}(h))_i &= \sum_{j \text{ s.t. } y_j = C_1} (\mat{K}_h)_{i,j}
    \end{align*}
    Yielding $\vecbf{Q}_{C_0}(h), \vecbf{Q}_{C_1}(h) \in \mathbb{R}^N$. $C_0 = \mathcal{C}[0], C_1 = \mathcal{C}[1]$.

    \item \textbf{Objective Score Calculation ($S(h)$):}
    \begin{itemize}
        \item If 'spatial': $S(h) = S_{spatial}(\vecbf{Q}_{C_0}(h), \vecbf{Q}_{C_1}(h), \vecbf{y})$.
        \item If 'axis': $S(h) = S_{axis}(\vecbf{Q}_{C_0}(h), \vecbf{Q}_{C_1}(h), \vecbf{y})$.
    \end{itemize}
    $S_{spatial}$ and $S_{axis}$ are externally defined objective functions.

    \item \textbf{Return Value:} $-S(h)$. If $S(h)$ is NaN or $K<2$, $\infty$ is returned.
\end{enumerate}

\subsection*{3. Model Training Procedure (\texttt{fit})}
\paragraph{Inputs:} $\mat{X} \in \mathbb{R}^{N \times D}$ (training data), $\vecbf{y} \in \mathcal{Y}^N$ (training labels).
\paragraph{Procedure:}
\begin{enumerate}
    \item Validate inputs. Store $\mat{X}_{train} = \mat{X}$. Identify unique classes $\mathcal{C}$.
    \item $(\eta, \mat{\Sigma}^{-1}) = \text{kernel\_fit}(\mat{X}_{train}, \text{type}=\text{kernel\_fit\_type})$.
    \item Optimize $h$:
    $$h_{opt} = \arg\max_{h \in [h_{min}, h_{max}]} S(h)$$
    This is achieved by minimizing $-S(h)$ using a scalar optimization routine.
    \item Compute final training kernel: $\mat{K}_{train\_opt} = \text{kernel}(\mat{X}_{train}, \mat{X}_{train}, \eta, \mat{\Sigma}^{-1}, h_{opt})$.
    \item Train SVM: Initialize $\text{SVC}(\text{kernel='precomputed'}, \dots)$. Fit with $\text{SVC.fit}(\mat{K}_{train\_opt}, \vecbf{y})$.
\end{enumerate}

\subsection*{4. Prediction Procedure (\texttt{predict})}
\paragraph{Inputs:} $\mat{X}_{new} \in \mathbb{R}^{M \times D}$ (new samples).
\paragraph{Procedure:}
\begin{enumerate}
    \item Validate $\mat{X}_{new}$.
    \item Compute test kernel matrix $\mat{K}_{test} \in \mathbb{R}^{M \times N}$:
    $$\mat{K}_{test} = \text{kernel}(\mat{X}_{new}, \mat{X}_{train}, \eta, \mat{\Sigma}^{-1}, h_{opt})$$
    \item Predict: $\hat{\vecbf{y}}_{new} = \text{SVC.predict}(\mat{K}_{test})$.
\end{enumerate}
\paragraph{Output:} $\hat{\vecbf{y}}_{new} \in \mathcal{Y}^M$ (predicted labels).
