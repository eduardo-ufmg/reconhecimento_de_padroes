\section{Resultados}

\subsection{Complexidade}
\begin{frame}
    \[ O(rk2^{\sqrt{p}}n\log{}n) \]
\end{frame}

\subsection{Sugestões de parâmetros}
\begin{frame}
\begin{itemize}
    \item $ k \in [1, 3] $
    \item $ m \approx \sqrt{p} $
    \item $ r < 1000 $
\end{itemize}
\end{frame}
